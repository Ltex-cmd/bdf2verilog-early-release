cmake_minimum_required(VERSION 3.16)
project(bdf2verilog LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(bdf STATIC
  src/parser.cpp src/schematic.cpp src/signal_name.cpp src/lpm.cpp
  src/verilog.cpp src/project.cpp src/bdf_tape.cpp src/bdf_span.cpp src/project_context.cpp)
target_include_directories(bdf PUBLIC include)
target_compile_definitions(bdf PUBLIC BDF_CONTEXT_TAPE)
add_executable(bdf-tool src/main.cpp)
target_link_libraries(bdf-tool PRIVATE bdf)
add_executable(benchmark-core tools/benchmark_core.cpp)
target_link_libraries(benchmark-core PRIVATE bdf)
include(CTest)
if(BUILD_TESTING)
  foreach(test IN ITEMS parser phase2 coordinate_numbers name_scope span_parser bus_contacts net_checks)
    add_executable(test-${test} tests/test_${test}.cpp)
    target_link_libraries(test-${test} PRIVATE bdf)
    add_test(NAME ${test} COMMAND test-${test})
  endforeach()
  add_executable(test-allocation tests/allocation_order.cpp)
  target_link_libraries(test-allocation PRIVATE bdf)
  add_test(NAME allocation COMMAND test-allocation
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/minimal.bdf ${CMAKE_CURRENT_BINARY_DIR}/allocation-minimal)
  add_test(NAME allocation-repeat COMMAND test-allocation
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/minimal.bdf ${CMAKE_CURRENT_BINARY_DIR}/allocation-minimal-repeat)
endif()
install(TARGETS bdf-tool RUNTIME DESTINATION bin)
install(FILES LICENSE README.md VERSION DESTINATION .)
